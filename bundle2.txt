"use client";

import { useRouter } from "next/navigation";
import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type ReactNode,
} from "react";
import { LOCALE_COOKIE, LOCALE_DIR, LOCALE_FORMAT, type Locale } from "@/i18n/config";
import { makeT, type TFunction } from "@/i18n/translate";

interface I18nValue {
  locale: Locale;
  dir: "ltr" | "rtl";
  t: TFunction;
  /** BCP-47 tag for Intl / toLocaleString. */
  formatTag: string;
  setLocale: (locale: Locale) => void;
}

const I18nContext = createContext<I18nValue | null>(null);

export function I18nProvider({
  initialLocale,
  children,
}: {
  initialLocale: Locale;
  children: ReactNode;
}) {
  const router = useRouter();
  const [locale, setLocaleState] = useState<Locale>(initialLocale);

  // Keep <html lang dir> in sync when the person switches language without a reload.
  useEffect(() => {
    document.documentElement.lang = locale;
    document.documentElement.dir = LOCALE_DIR[locale];
  }, [locale]);

  const setLocale = useCallback(
    (next: Locale) => {
      setLocaleState(next);
      // One year. Read by the server so the first paint already uses the right language.
      document.cookie = `${LOCALE_COOKIE}=${next}; path=/; max-age=31536000; samesite=lax`;
      router.refresh(); // re-render server components in the new language
    },
    [router],
  );

  const value = useMemo<I18nValue>(
    () => ({
      locale,
      dir: LOCALE_DIR[locale],
      t: makeT(locale),
      formatTag: LOCALE_FORMAT[locale],
      setLocale,
    }),
    [locale, setLocale],
  );

  return <I18nContext.Provider value={value}>{children}</I18nContext.Provider>;
}

export function useI18n(): I18nValue {
  const ctx = useContext(I18nContext);
  if (!ctx) throw new Error("useI18n must be used within I18nProvider");
  return ctx;
}

export function useT(): TFunction {
  return useI18n().t;
}

export type Locale = (typeof LOCALES)[number];

export const DEFAULT_LOCALE: Locale = "en";
export const LOCALE_COOKIE = "cova-locale";

/** Names are written in their own language so people can always find theirs. */
export const LOCALE_NAMES: Record<Locale, string> = {
  en: "English",
  fr: "FranÃ§ais",
  ar: "Ø§Ù„Ø¹Ø±Ø¨ÙŠØ©",
};

export const LOCALE_DIR: Record<Locale, "ltr" | "rtl"> = {
  en: "ltr",
  fr: "ltr",
  ar: "rtl",
};

/** BCP-47 tags for Intl formatting (Arabic uses Latin digits to match the rest of the UI). */
export const LOCALE_FORMAT: Record<Locale, string> = {
  en: "en-US",
  fr: "fr-FR",
  ar: "ar-MA",
};

export function isLocale(value: string | undefined | null): value is Locale {
  return !!value && (LOCALES as readonly string[]).includes(value);
}

/** Picks the best supported locale from an Accept-Language header. */
export function matchAcceptLanguage(header: string | null | undefined): Locale {
  if (!header) return DEFAULT_LOCALE;
  const wanted = header
    .split(",")
    .map((part) => part.trim().split(";")[0]?.toLowerCase().split("-")[0] ?? "")
    .filter(Boolean);
  for (const tag of wanted) if (isLocale(tag)) return tag;
  return DEFAULT_LOCALE;
}

import { cookies, headers } from "next/headers";
import { LOCALE_COOKIE, isLocale, matchAcceptLanguage, type Locale } from "@/i18n/config";
import { makeT, type TFunction } from "@/i18n/translate";

/** Locale for the current request: saved choice first, then the browser's language. */
export async function getLocale(): Promise<Locale> {
  const saved = (await cookies()).get(LOCALE_COOKIE)?.value;
  if (isLocale(saved)) return saved;
  return matchAcceptLanguage((await headers()).get("accept-language"));
}

export async function getT(): Promise<TFunction> {
  return makeT(await getLocale());
}

import { en, type MessageKey } from "@/i18n/messages/en";
import { fr } from "@/i18n/messages/fr";
import { ar } from "@/i18n/messages/ar";
import type { Locale } from "@/i18n/config";

export type { MessageKey };
export type Vars = Record<string, string | number>;
export type TFunction = (key: MessageKey, vars?: Vars) => string;

const catalogs: Record<Locale, Record<MessageKey, string>> = { en, fr, ar };

/** Looks up a message, falling back to English, then to the key itself. Replaces {name} placeholders. */
export function translate(locale: Locale, key: MessageKey, vars?: Vars): string {
  const template = catalogs[locale][key] ?? en[key] ?? key;
  if (!vars) return template;
  return template.replace(/\{(\w+)\}/g, (match, name: string) =>
    name in vars ? String(vars[name]) : match,
  );
}

export function makeT(locale: Locale): TFunction {
  return (key, vars) => translate(locale, key, vars);
}

"use client";

import { useState, type FormEvent } from "react";
import { Button } from "@/components/ui/Button";
import { Card, CardBody, CardHeader } from "@/components/ui/Card";
import { DemoNotice } from "@/components/ui/DemoNotice";
import { SelectFilter, TextAreaField, TextField } from "@/components/ui/FormField";
import { LanguageSwitcher } from "@/components/ui/LanguageSwitcher";
import { Switch } from "@/components/ui/Switch";
import { useToast } from "@/components/ui/Toast";
import { useT } from "@/i18n/client";
import type { MessageKey } from "@/i18n/translate";
import { demoActions, useDemoState, useHydrated } from "@/lib/demo-store";
import type { UserSettings } from "@/types";

const visibilityKeys: Record<UserSettings["profileVisibility"], MessageKey> = {
  everyone: "settings.vis.everyone",
  groups: "settings.vis.groups",
  private: "settings.vis.private",
};

function SettingsFormInner({ initial }: { initial: UserSettings }) {
  const toast = useToast();
  const t = useT();
  const visibilityLabels = Object.fromEntries(
    Object.entries(visibilityKeys).map(([k, key]) => [k, t(key)]),
  ) as Record<UserSettings["profileVisibility"], string>;
  const [values, setValues] = useState(initial);
  const [errors, setErrors] = useState<{ displayName?: string; email?: string }>({});
  const set = <K extends keyof UserSettings>(key: K, value: UserSettings[K]) =>
    setValues((v) => ({ ...v, [key]: value }));

  function onSubmit(e: FormEvent) {
    e.preventDefault();
    const found: typeof errors = {};
    if (values.displayName.trim().length < 2) found.displayName = t("settings.err.displayName");
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(values.email.trim()))
      found.email = t("settings.err.email");
    setErrors(found);
    if (Object.keys(found).length > 0) return;
    demoActions.saveSettings({
      ...values,
      displayName: values.displayName.trim(),
      email: values.email.trim(),
    });
    toast.push(t("settings.saved"));
  }

  return (
    <form onSubmit={onSubmit} noValidate className="max-w-2xl space-y-6">
      <Card>
        <CardHeader title={t("settings.profile")} description={t("settings.profileDesc")} />
        <CardBody className="space-y-4">
          <TextField
            id="displayName"
            label={t("settings.displayName")}
            value={values.displayName}
            error={errors.displayName}
            onChange={(e) => set("displayName", e.target.value)}
          />
          <TextField
            id="settings-email"
            label={t("settings.email")}
            type="email"
            value={values.email}
            error={errors.email}
            onChange={(e) => set("email", e.target.value)}
          />
          <TextAreaField
            id="bio"
            label={t("settings.bio")}
            value={values.bio}
            onChange={(e) => set("bio", e.target.value)}
          />
        </CardBody>
      </Card>

      <Card>
        <CardHeader title={t("lang.label")} description={t("lang.description")} />
        <CardBody>
          <LanguageSwitcher />
        </CardBody>
      </Card>

      <Card>
        <CardHeader title={t("settings.notifications")} />
        <CardBody className="divide-y divide-cova-line py-2">
          <Switch
            label={t("settings.emailNotif")}
            description={t("settings.emailNotifDesc")}
            checked={values.emailNotifications}
            onChange={(v) => set("emailNotifications", v)}
          />
          <Switch
            label={t("settings.messageAlerts")}
            description={t("settings.messageAlertsDesc")}
            checked={values.messageAlerts}
            onChange={(v) => set("messageAlerts", v)}
          />
          <Switch
            label={t("settings.digest")}
            description={t("settings.digestDesc")}
            checked={values.weeklyDigest}
            onChange={(v) => set("weeklyDigest", v)}
          />
        </CardBody>
      </Card>

      <Card>
        <CardHeader title={t("settings.privacy")} />
        <CardBody>
          <SelectFilter
            label={t("settings.visibility")}
            options={Object.values(visibilityLabels)}
            value={visibilityLabels[values.profileVisibility]}
            onChange={(e) => {
              const key = (Object.keys(visibilityKeys) as UserSettings["profileVisibility"][]).find(
                (k) => visibilityLabels[k] === e.target.value,
              );
              if (key) set("profileVisibility", key);
            }}
          />
        </CardBody>
      </Card>

      <DemoNotice>{t("settings.demoNotice")}</DemoNotice>
      <div className="flex gap-2">
        <Button type="submit">{t("common.save")}</Button>
        <Button
          variant="secondary"
          onClick={() => {
            setValues(initial);
            setErrors({});
          }}
        >
          {t("common.discard")}
        </Button>
      </div>
    </form>
  );
}

export function SettingsForm() {
  const { settings } = useDemoState();
  const hydrated = useHydrated();
  // Re-mount once the saved browser state has loaded so the form starts from saved values.
  return <SettingsFormInner key={hydrated ? "ready" : "loading"} initial={settings} />;
}
